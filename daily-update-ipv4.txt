name: Daily Fetch CF IPs

on:
  schedule:
    #  UTC时间 20:00，相当于北京时间每天早上 04:00 运行一次
    - cron: '0 20 * * *'
  workflow_dispatch: # 支持手动触发

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Fetch and Filter IPs
        run: |
          # 1. 获取原项目IPv4，随机抽取或取前80个（以80个为例）
          curl -s https://raw.githubusercontent.com/LancelotRar/best-cf-ips/main/best-cf-ipv4.txt | head -n 80 > best-cf-ipv4.txt
          
          # 2. 获取其他源的IPv6，并截取前80个
          curl -s https://raw.githubusercontent.com/cmliu/CFnat-500in1/main/bestcf-ipv6.txt | head -n 80 > best-cf-ipv6.txt

      - name: Commit and Push
        run: |
          git config --global user.name "github-actions[bot]"
          git config --global user.email "github-actions[bot]@users.noreply.github.com"
          git add best-cf-ipv4.txt best-cf-ipv6.txt
          git commit -m "Auto update best CF IPs" || exit 0
          git push
